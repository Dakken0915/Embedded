# 网络编程中的多连接处理

TCP 服务端通常由监听套接字接收连接，再用各自的连接套接字收发数据。多个连接如何推进，取决于线程/进程模型与 I/O 模型。TCP 字节流与基本接口见 [Socket 与 I/O 模型](./Socket与IO模型.md)。

## 每连接一个线程或进程

主线程或主进程循环调用 `accept()`，把新连接交给工作线程或子进程，各自阻塞读写。线程应使用 `pthread_create` 或 C++ `std::thread` 等接口创建；`fork()` 创建进程，不创建线程。

| 模型 | 特点 | 资源与生命周期 |
| --- | --- | --- |
| 每连接一个线程 | 控制流直接，线程共享地址空间 | 线程栈、调度与切换有成本；共享状态要同步，结束后 join 或 detach |
| 每连接一个进程 | 地址空间隔离，可独立阻塞处理 | 父子进程关闭不使用的 fd，父进程回收子进程，处理 fork 失败与数量上限 |
| 线程池/进程池 | 复用固定或有界的工作单元 | 任务队列要有容量、超时与过载策略；连接长期阻塞仍会占用工作单元 |

单个连接阻塞时，其他工作单元可以继续运行，但仍会竞争 CPU、内存、锁和描述符；不能认为彼此完全不受影响。多线程进程中 `fork()` 的子进程在 `exec` 前还受到异步信号安全限制，不能直接按普通线程代码继续执行复杂业务。

## I/O 多路复用

事件循环通过 `select`、`poll` 或 Linux `epoll` 等待一组 fd 的就绪事件，再推进对应连接的状态。就绪通知不等于完成了读写，也不代表已经收到完整应用报文。

| 接口 | 就绪集合的表达 | 边界 |
| --- | --- | --- |
| `select` | 每次传入 fd 位集合，返回后检查集合 | 常见 libc 的 `fd_set` 受 `FD_SETSIZE` 限制，集合每轮需重建 |
| `poll` | 每次传入 `pollfd` 数组 | 不受上述位集合限制，仍要处理传入数组及结果 |
| `epoll` | `epoll_ctl` 维护关注集合，`epoll_wait` 返回就绪项 | Linux 专用；适合大量连接的常见方案，收益取决于活跃比例与业务负载 |

新代码通常使用 `epoll_create1(EPOLL_CLOEXEC)`；旧 `epoll_create(1)` 的参数不是容量上限，现代 Linux 已忽略这个大小提示，但仍要求大于零。[epoll_create(2)](https://man7.org/linux/man-pages/man2/epoll_create.2.html)

事件循环的典型路径为：监听 fd 可读时接受连接并登记；连接可读时收取字节并解析协议；有待发字节时关注可写事件；发送完成后撤销可写关注；EOF 或错误时清理状态和 fd。单循环中的耗时计算和阻塞库调用会拖住全部连接，可把这类工作交给有界工作池，由事件循环统一管理连接生命周期。

## 非阻塞与触发方式

Linux `accept()` 返回的新 socket 不继承监听 socket 的 `O_NONBLOCK`；可用 `accept4(..., SOCK_NONBLOCK | SOCK_CLOEXEC)` 原子设置属性，见 [accept(2)](https://man7.org/linux/man-pages/man2/accept.2.html)。

`epoll` 默认采用水平触发（LT），就绪条件持续存在时可以再次通知；边沿触发（ET）使用 `EPOLLET`，一般需要非阻塞操作并持续读取、接受或发送到 `EAGAIN/EWOULDBLOCK`。若因公平性预算或应用缓冲容量提前停止 ET 处理，须安排主动继续处理，不能假设还会出现新边沿。[epoll(7)](https://www.man7.org/linux/man-pages/man7/epoll.7.html)

非阻塞 `recv` 的正返回值表示本次字节数，0 表示对端发送方向有序关闭，负返回值应区分 `EINTR`、暂不可用和实际错误。`send` 可能短写，须保存偏移；只有存在待发字节时才关注 `EPOLLOUT`，否则可写 socket 可能使循环空转。`MSG_NOSIGNAL` 抑制本次发送的 SIGPIPE，错误仍须处理。[recv(2)](https://man7.org/linux/man-pages/man2/recv.2.html)、[send(2)](https://man7.org/linux/man-pages/man2/send.2.html)

## 有界的 LT epoll 回显示例

下面是 Linux/glibc、C11 完整程序，依赖 POSIX socket 与 Linux epoll/accept4，监听 `127.0.0.1:9090`。最多保留 128 个客户端，每个连接只有一个 4096 字节缓冲；发送完这一批再读下一批，因此慢接收端不会导致用户态待发队列无限增长。每次连接事件只执行一次读或写，监听事件也有接受预算，借助 LT 后续通知继续推进。

```c
#define _GNU_SOURCE
#include <arpa/inet.h>
#include <errno.h>
#include <signal.h>
#include <stdint.h>
#include <stdio.h>
#include <sys/epoll.h>
#include <sys/socket.h>
#include <unistd.h>

enum { MAX_CLIENTS = 128, BUFFER_SIZE = 4096, MAX_EVENTS = 64 };
struct client {
    int fd;
    size_t used, sent;
    unsigned char buffer[BUFFER_SIZE];
};
static struct client clients[MAX_CLIENTS];
static volatile sig_atomic_t stopping;

static void stop_server(int signo)
{
    (void)signo;
    stopping = 1;
}

static void close_client(int epfd, unsigned slot)
{
    (void)epoll_ctl(epfd, EPOLL_CTL_DEL, clients[slot].fd, NULL);
    (void)close(clients[slot].fd); /* Linux 上不因 EINTR 重试 close */
    clients[slot].fd = -1;
}

static int accept_clients(int epfd, int listener)
{
    for (unsigned budget = 0; budget < 32; ++budget) {
        int fd = accept4(listener, NULL, NULL,
                         SOCK_NONBLOCK | SOCK_CLOEXEC);
        if (fd < 0) {
            if (errno == EINTR || errno == ECONNABORTED) continue;
            if (errno == EAGAIN || errno == EWOULDBLOCK) return 0;
            return -1; /* 包括资源耗尽：交给主循环统一退出清理 */
        }
        unsigned slot = 0;
        while (slot < MAX_CLIENTS && clients[slot].fd >= 0) ++slot;
        if (slot == MAX_CLIENTS) {
            (void)close(fd); /* 明确的连接数上限 */
            continue;
        }
        struct epoll_event event = { .events = EPOLLIN };
        event.data.u32 = slot;
        if (epoll_ctl(epfd, EPOLL_CTL_ADD, fd, &event) < 0) {
            int saved = errno;
            (void)close(fd);
            errno = saved;
            return -1;
        }
        clients[slot].fd = fd;
        clients[slot].used = clients[slot].sent = 0;
    }
    return 0;
}

static void service_client(int epfd, unsigned slot, uint32_t flags)
{
    struct client *c = &clients[slot];
    if (flags & EPOLLERR) {
        close_client(epfd, slot);
        return;
    }
    if (c->sent < c->used) {
        ssize_t n = send(c->fd, c->buffer + c->sent,
                         c->used - c->sent, MSG_NOSIGNAL);
        if (n > 0) c->sent += (size_t)n;
        else if (n == 0 || (errno != EINTR && errno != EAGAIN &&
                            errno != EWOULDBLOCK)) goto closed;
    } else {
        ssize_t n = recv(c->fd, c->buffer, sizeof c->buffer, 0);
        if (n > 0) {
            c->used = (size_t)n;
            c->sent = 0;
        } else if (n == 0 || (errno != EINTR && errno != EAGAIN &&
                              errno != EWOULDBLOCK)) goto closed;
    }
    struct epoll_event event = {
        .events = c->sent < c->used ? EPOLLOUT : EPOLLIN
    };
    event.data.u32 = slot;
    if (epoll_ctl(epfd, EPOLL_CTL_MOD, c->fd, &event) == 0) return;
closed:
    close_client(epfd, slot);
}

int main(void)
{
    int listener = -1, epfd = -1, status = 1;
    for (unsigned i = 0; i < MAX_CLIENTS; ++i) clients[i].fd = -1;
    struct sigaction action = { .sa_handler = stop_server };
    if (sigemptyset(&action.sa_mask) < 0 ||
        sigaction(SIGINT, &action, NULL) < 0 ||
        sigaction(SIGTERM, &action, NULL) < 0) goto failed;

    listener = socket(AF_INET, SOCK_STREAM | SOCK_NONBLOCK | SOCK_CLOEXEC, 0);
    if (listener < 0) goto failed;
    int enabled = 1;
    if (setsockopt(listener, SOL_SOCKET, SO_REUSEADDR,
                   &enabled, sizeof enabled) < 0) goto failed;
    struct sockaddr_in address = {
        .sin_family = AF_INET,
        .sin_port = htons(9090),
        .sin_addr = { .s_addr = htonl(INADDR_LOOPBACK) }
    };
    if (bind(listener, (struct sockaddr *)&address, sizeof address) < 0 ||
        listen(listener, 128) < 0) goto failed;
    epfd = epoll_create1(EPOLL_CLOEXEC);
    if (epfd < 0) goto failed;
    struct epoll_event event = { .events = EPOLLIN };
    event.data.u32 = UINT32_MAX; /* 与客户端槽位区分 */
    if (epoll_ctl(epfd, EPOLL_CTL_ADD, listener, &event) < 0) goto failed;

    while (!stopping) {
        struct epoll_event events[MAX_EVENTS];
        /* 有限超时也覆盖检查停止标志与进入等待之间的信号窗口。 */
        int count = epoll_wait(epfd, events, MAX_EVENTS, 1000);
        if (count < 0) {
            if (errno == EINTR) continue;
            goto failed;
        }
        for (int i = 0; i < count; ++i) {
            uint32_t slot = events[i].data.u32;
            if (slot == UINT32_MAX) {
                if (events[i].events & (EPOLLERR | EPOLLHUP)) {
                    errno = EIO;
                    goto failed;
                }
                if (accept_clients(epfd, listener) < 0) goto failed;
            } else if (slot < MAX_CLIENTS && clients[slot].fd >= 0) {
                service_client(epfd, slot, events[i].events);
            }
        }
    }
    status = 0;
    goto cleanup;
failed:
    perror("epoll server");
cleanup:
    for (unsigned i = 0; i < MAX_CLIENTS; ++i) {
        if (clients[i].fd >= 0) close_client(epfd, i);
    }
    if (epfd >= 0) (void)close(epfd);
    if (listener >= 0) (void)close(listener);
    return status;
}
```

保存为 `epoll-server.c`，使用 `cc -std=c11 -Wall -Wextra -Werror epoll-server.c -o epoll-server` 编译。运行后可从多个终端连接回环地址测试，SIGINT 或 SIGTERM 使循环退出并统一清理；示例停止时直接关闭连接，不承诺排空全部待发数据。

这里不因 `EPOLLHUP` 直接丢弃缓冲区，而是继续非阻塞读写，最终依据 EOF 或实际错误关闭。对端 `shutdown(SHUT_WR)` 后仍可读取回显，服务端会先发送已读字节，再读取到 EOF。`EPOLLERR` 则按失败连接立即清理。

该程序是字节回显服务，没有应用层报文边界、认证或业务任务。完整服务还需增加空闲/请求超时、连接配额与明确的过载处理；慢客户端虽然不能撑大单连接用户缓冲，仍可长期占用连接槽位。若改为多线程共享连接或异步任务持有 fd，还须防止关闭后 fd 复用及失效事件引用，不能原样沿用此处单线程所有权假设。
